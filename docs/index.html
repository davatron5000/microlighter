Go

package highlight

import "fmt"

// Token is a scoped span of source text.
type Token struct {
	Scope      string
	Start, End int
}

// Tokenize splits source into scoped tokens.
func Tokenize(source string, scopes []string) ([]Token, error) {
	if len(scopes) == 0 {
		return nil, fmt.Errorf("no scopes for %q", source)
	}

	tokens := make([]Token, 0, len(scopes))
	for i, scope := range scopes {
		tokens = append(tokens, Token{Scope: scope, Start: i})
	}
	return tokens, nil
}

# MicroLighter

A tiny, **dependency-free** syntax highlighter built on the
[CSS Custom Highlight API](https://developer.mozilla.org/).

## Install

    npm install microlighter

Highlights *without* `<span>` soup — ranges live in `CSS.highlights`,
and grammars are lazy-loaded on demand.

> Your markup stays clean; the color lives in CSS.

from dataclasses import dataclass


@dataclass
class Token:
    """A highlighted range with a scope."""

    scope: str
    start: int
    end: int = 0


def tokenize(source: str, scopes: list[str]) -> list[Token]:
    tokens = [Token(scope, i) for i, scope in enumerate(scopes)]
    if not tokens:
        raise ValueError("no scopes")
    return sorted(tokens, key=lambda t: t.start)


if __name__ == "__main__":
    for token in tokenize("const x = 42", ["keyword", "number"]):
        print(f"{token.scope}: {token.start}")

Rust

use std::collections::HashMap;

/// A scoped span of source text.
#[derive(Debug, Clone)]
struct Token<'a> {
    scope: &'a str,
    start: usize,
    end: usize,
}

fn tokenize<'a>(source: &'a str, scopes: &[&'a str]) -> Result<Vec<Token<'a>>, String> {
    if scopes.is_empty() {
        return Err(format!("no scopes for {source:?}"));
    }

    let mut counts: HashMap<&str, u32> = HashMap::new();
    let tokens = scopes
        .iter()
        .enumerate()
        .map(|(i, &scope)| {
            *counts.entry(scope).or_insert(0) += 1;
            Token { scope, start: i, end: i + 1 }
        })
        .collect();
    Ok(tokens)
}

import { highlightAll } from "microlighter";

/** A scoped span of source text. */
interface Token<T = string> {
  readonly scope: T;
  start: number;
  end?: number;
}

type Scopes = ReadonlyMap<string, number>;

enum Kind {
  Keyword,
  String,
}

export class Tokenizer implements Iterable<Token> {
  #tokens: Token[] = [];

  constructor(private readonly source: string) {}

  tokenize(scopes: readonly string[]): Token[] {
    return scopes
      .map((scope, i): Token => ({ scope, start: i }))
      .filter((token) => token.start >= 0);
  }

  *[Symbol.iterator](): Iterator<Token> {
    yield* this.#tokens;
  }
}
